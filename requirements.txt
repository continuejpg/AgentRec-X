# Runtime dependencies for the AgentRec-X recommendation service (Milestone 6) and
# the minimal LangGraph agent (Milestone 7B).
#
#   .venv/bin/python -m pip install -r requirements.txt
#
# The recommendation library itself (preprocessing, evaluation, baselines, datasets,
# models, training, inference, tools) imports the standard library, NumPy and PyTorch.
# NumPy is a *direct* import on the served path (recommendation/inference/ranking.py),
# so it is pinned below rather than left to resolve transitively through PyTorch.
#
# PyTorch is intentionally NOT listed here: it is environment-specific, and AGENTS.md
# says do not reinstall or upgrade the CUDA/PyTorch stack.  For a fresh CPU-only
# environment use the dedicated, index-pinned file instead:
#
#   .venv/bin/python -m pip install -r requirements-cpu.txt
#
# A plain `pip install torch` is NOT equivalent -- it pulls the PyPI CUDA wheel and
# its `nvidia-*` dependencies onto a machine with no GPU.  See requirements-cpu.txt.
#
# NOTE: pytest-asyncio lives in requirements-dev.txt; the API tests use FastAPI's
# synchronous TestClient, which needs httpx (already present).
#
# After installing, `.venv/bin/python -m pip check` must report
# "No broken requirements found."

# --- Milestone 6: HTTP recommendation service --------------------------------- #
fastapi>=0.110,<1.0
uvicorn>=0.27,<1.0

# NumPy is imported directly by the serving ranking code (see the header note).  The
# acceptance-pinned version is listed explicitly so the ranking contract cannot drift
# with whatever NumPy a future PyTorch happens to require.
numpy==1.26.4

# Pydantic is FastAPI's validation core; it is pinned here indirectly but listed
# explicitly so the served contract does not drift silently.
pydantic>=2.5,<3.0

# --- Milestone 7B: minimal agent orchestration -------------------------------- #
# Exactly one framework is added, and it is used for orchestration only.  No LLM
# provider SDK is required: the milestone's decision model is injected (see
# recommendation/agent/README.md), so `openai`, `anthropic`, `google-generativeai`
# and friends are deliberately absent, no API key is read and no network call is
# made.  Only `langgraph.graph` is imported by our code.
langgraph>=1.0,<2.0

# ----------------------------------------------------------------------------- #
# REQUIRED COMPATIBILITY PIN -- do not drop without re-running `pip check`
# ----------------------------------------------------------------------------- #
# `langsmith` is a transitive dependency of `langgraph` (via `langchain-core`, which
# allows `langsmith<1.0.0,>=0.3.45`).  Pinning it below 0.12 is REQUIRED, not
# cosmetic:
#
#   * langsmith <0.12  declares `httpx<1,>=0.23.0`
#   * langsmith >=0.12 declares `httpx2<3,>=2`
#
# `httpx2` pulls `httpcore2`, which requires `h11>=0.16`.  The accepted Milestone 6
# HTTP stack is `httpx==0.27.0` -> `httpcore==1.0.5` -> `h11<0.15`, so installing
# httpx2 makes `pip check` fail and breaks the FastAPI/TestClient stack.  Capping
# langsmith here keeps the whole closure satisfiable on the accepted h11 0.14.0.
#
# LangSmith is LangChain's optional remote tracing client: it is not imported by
# `recommendation/agent/`, and tracing is never enabled.  Verified working pair
# (Python 3.10.8, `pip check` clean): langgraph 1.2.11 / langchain-core 1.6.3 /
# langsmith 0.11.2 / httpx 0.27.0 / httpcore 1.0.5 / h11 0.14.0.
langsmith>=0.3.45,<0.12
