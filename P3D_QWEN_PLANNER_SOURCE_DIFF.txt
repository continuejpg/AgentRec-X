+"""Minimal opt-in Planner Action Contract for AgentRec-X P3D.
+
+This is a contract/orchestration prototype, not a new recommender.  Actions are
+validated and executed against the existing DemoAgentService (M5/R1/M6.1 and the
+CandidateLedger are inside that service); the planner only chooses the next action.
+
+The built-in ``StubPlanner`` is a deterministic CPU test double.  It is not Qwen and
+must not be reported as model reasoning.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass, field, replace
+from enum import Enum
+from typing import Any, Protocol, Sequence
+
+
+class PlannerAction(str, Enum):
+    ASK = "ASK"
+    RETRIEVE = "RETRIEVE"
+    RERANK = "RERANK"
+    FINALIZE = "FINALIZE"
+
+
+@dataclass(frozen=True)
+class PlannerObservation:
+    step: int
+    query: str
+    candidate_ids: tuple[str, ...] = ()
+    state_version: str | None = None
+    needs_user_input: bool = False
+    reranked: bool = False
+    terminal: bool = False
+    note: str = ""
+    #: Dense, read-only context for an LLM planner.  It is never used as a candidate
+    #: source and never bypasses role/price/catalogue validation.
+    m5_state: dict[str, Any] = field(default_factory=dict)
+    allowed_actions: tuple[str, ...] = ()
+    last_tool_observation: dict[str, Any] = field(default_factory=dict)
+    remaining_steps: int = 0
+
+
+@dataclass(frozen=True)
+class ActionProposal:
+    action: PlannerAction
+    arguments: dict[str, Any] = field(default_factory=dict)
+    rationale: str = ""
+
+
+class PlannerContractError(RuntimeError):
+    def __init__(self, code: str, detail: str) -> None:
+        super().__init__(detail)
+        self.code = code
+        self.detail = detail
+
+
+def allowed_planner_actions(observation: PlannerObservation) -> tuple[str, ...]:
+    if observation.terminal:
+        return ()
+    actions = [PlannerAction.ASK.value, PlannerAction.RETRIEVE.value]
+    if observation.candidate_ids:
+        actions.extend((PlannerAction.RERANK.value, PlannerAction.FINALIZE.value))
+    return tuple(actions)
+
+
+def validate_action_arguments(proposal: ActionProposal) -> None:
+    if proposal.action is PlannerAction.ASK:
+        if not str(proposal.arguments.get("question") or "").strip():
+            raise PlannerContractError("ask_missing_question", "ASK requires a non-empty question")
+    if proposal.action is PlannerAction.RETRIEVE:
+        if not str(proposal.arguments.get("query") or "").strip():
+            raise PlannerContractError("retrieve_missing_query", "RETRIEVE requires a non-empty query")
+
+
+def validate_action(
+    proposal: ActionProposal,
+    observation: PlannerObservation,
+    *,
+    step: int,
+    max_steps: int,
+) -> None:
+    validate_action_arguments(proposal)
+    if step > max_steps:
+        raise PlannerContractError("max_steps_exceeded", f"step {step} > {max_steps}")
+    if observation.terminal:
+        raise PlannerContractError("terminal_observation", "no action after terminal observation")
+    if proposal.action is PlannerAction.FINALIZE and not observation.candidate_ids:
+        raise PlannerContractError("finalize_without_candidates", "cannot finalize an empty candidate set")
+    if proposal.action is PlannerAction.RERANK and not observation.candidate_ids:
+        raise PlannerContractError("rerank_without_candidates", "cannot rerank an empty candidate set")
+    if proposal.action is PlannerAction.ASK and observation.candidate_ids and not observation.needs_user_input:
+        raise PlannerContractError("ask_without_need", "ASK requires an ambiguity/need signal")
+    if proposal.action is PlannerAction.RETRIEVE and observation.candidate_ids and observation.reranked:
+        raise PlannerContractError("retrieve_after_rerank", "retrieve after a completed rerank is not legal")
+
+
+class StubPlanner:
+    """Deterministic CPU planner used only for the prototype trace."""
+
+    def propose(self, observation: PlannerObservation) -> ActionProposal:
+        if observation.terminal:
+            raise PlannerContractError("stub_terminal", "stub planner received terminal observation")
+        if observation.needs_user_input:
+            return ActionProposal(PlannerAction.ASK, {"question": "Could you clarify the product type?"}, "ambiguity signal")
+        if not observation.candidate_ids:
+            return ActionProposal(PlannerAction.RETRIEVE, {"query": observation.query}, "no candidates yet")
+        if not observation.reranked:
+            return ActionProposal(PlannerAction.RERANK, {"candidates": list(observation.candidate_ids)}, "existing M6.1 rerank")
+        return ActionProposal(PlannerAction.FINALIZE, {"candidates": list(observation.candidate_ids)}, "candidate set is ready")
+
+
+@dataclass(frozen=True)
+class PlannerStep:
+    step: int
+    action: str
+    tool: str
+    action_arguments: dict[str, Any]
+    observation_before: dict[str, Any]
+    tool_observation: dict[str, Any]
+    observation_after: dict[str, Any]
+    state_version: str | None
+    terminal: bool
+
+
+@dataclass
+class PlannerRun:
+    steps: list[PlannerStep] = field(default_factory=list)
+    final_ids: tuple[str, ...] = ()
+    termination: str = ""
+
+    def as_dict(self) -> dict[str, Any]:
+        return {
+            "steps": [
+                {
+                    "step": step.step,
+                    "action": step.action,
+                    "tool": step.tool,
+                    "action_arguments": step.action_arguments,
+                    "observation_before": step.observation_before,
+                    "tool_observation": step.tool_observation,
+                    "observation_after": step.observation_after,
+                    "state_version": step.state_version,
+                    "terminal": step.terminal,
+                }
+                for step in self.steps
+            ],
+            "final_ids": list(self.final_ids),
+            "termination": self.termination,
+        }
+
+
+class PlannerExecutor(Protocol):
+    def execute(self, proposal: ActionProposal, observation: PlannerObservation) -> tuple[PlannerObservation, dict[str, Any], bool]: ...
+
+
+def run_planner(
+    *,
+    planner: Any,
+    executor: PlannerExecutor,
+    initial_observation: PlannerObservation,
+    max_steps: int = 4,
+) -> PlannerRun:
+    observation = initial_observation
+    run = PlannerRun()
+    for step in range(1, max_steps + 1):
+        observation = replace(
+            observation,
+            allowed_actions=allowed_planner_actions(observation),
+            remaining_steps=max(0, max_steps - step),
+        )
+        proposal = planner.propose(observation)
+        if proposal.action.value not in observation.allowed_actions:
+            raise PlannerContractError("action_not_allowed", f"{proposal.action.value} is not allowed now")
+        validate_action(proposal, observation, step=step, max_steps=max_steps)
+        after, tool_observation, terminal = executor.execute(proposal, observation)
+        after = replace(
+            after,
+            last_tool_observation=dict(tool_observation),
+            remaining_steps=max(0, max_steps - step),
+        )
+        run.steps.append(
+            PlannerStep(
+                step=step,
+                action=proposal.action.value,
+                tool=proposal.action.value.lower(),
+                action_arguments=dict(proposal.arguments),
+                observation_before={
+                    "step": observation.step,
+                    "candidate_ids": list(observation.candidate_ids),
+                    "state_version": observation.state_version,
+                    "needs_user_input": observation.needs_user_input,
+                    "reranked": observation.reranked,
+                },
+                tool_observation=tool_observation,
+                observation_after={
+                    "step": after.step,
+                    "candidate_ids": list(after.candidate_ids),
+                    "state_version": after.state_version,
+                    "needs_user_input": after.needs_user_input,
+                    "reranked": after.reranked,
+                    "terminal": after.terminal,
+                },
+                state_version=after.state_version,
+                terminal=terminal,
+            )
+        )
+        observation = after
+        if terminal:
+            run.termination = proposal.action.value
+            run.final_ids = tuple(after.candidate_ids)
+            break
+    else:
+        run.termination = "max_steps"
+        run.final_ids = tuple(observation.candidate_ids)
+    return run

+"""CPU executor for the P3D planner contract.
+
+The executor reuses DemoAgentService.run_shopping_turn for RETRIEVE, the service's
+existing M5/M6.1 private path for RERANK, and returns the service's already validated
+recommendations for FINALIZE.  It never creates candidates, bypasses role checks or
+bypasses price constraints.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass
+from typing import Any
+
+from recommendation.demo.agent_schemas import AgentRecommendRequest
+from recommendation.demo.planner_contract import (
+    ActionProposal,
+    PlannerAction,
+    PlannerObservation,
+)
+
+
+@dataclass
+class ServicePlannerExecutor:
+    service: Any
+    payload: AgentRecommendRequest
+    session_id: str
+    task_id: str
+    turn_prefix: str = "p3d"
+    _candidates: tuple[Any, ...] = ()
+    _turn: int = 0
+
+    def _observation(self, *, candidate_ids: tuple[str, ...], note: str = "", reranked: bool = False, terminal: bool = False, needs_user_input: bool = False) -> PlannerObservation:
+        diagnostics = self.service.state_diagnostics(self.task_id) or {}
+        return PlannerObservation(
+            step=diagnostics.get("turn_count", self._turn),
+            query=self.payload.message,
+            candidate_ids=candidate_ids,
+            state_version=str(diagnostics.get("state_digest") or diagnostics.get("state_version") or ""),
+            needs_user_input=needs_user_input,
+            reranked=reranked,
+            terminal=terminal,
+            note=note,
+            m5_state={
+                "mode": diagnostics.get("mode"),
+                "event_count": diagnostics.get("event_count"),
+                "preference_count": diagnostics.get("preference_count"),
+                "hard_constraint_count": diagnostics.get("hard_constraint_count"),
+                "constraint_kinds": diagnostics.get("constraint_kinds", ()),
+                "ranker_feature_count": diagnostics.get("ranker_feature_count"),
+                "state_digest": diagnostics.get("state_digest"),
+            },
+        )
+
+    def execute(self, proposal: ActionProposal, observation: PlannerObservation) -> tuple[PlannerObservation, dict[str, Any], bool]:
+        if proposal.action is PlannerAction.ASK:
+            return (
+                self._observation(candidate_ids=observation.candidate_ids, needs_user_input=True, note="awaiting user"),
+                {"tool": "ask", "question": proposal.arguments.get("question", "")},
+                False,
+            )
+        if proposal.action is PlannerAction.RETRIEVE:
+            self._turn += 1
+            response = self.service.run_shopping_turn(
+                self.payload.model_copy(update={"message": str(proposal.arguments.get("query") or self.payload.message)}),
+                session_id=self.session_id,
+                task_id=self.task_id,
+                turn_id=f"{self.turn_prefix}:{self._turn:03d}",
+            )
+            self._candidates = tuple(response.recommendations)
+            ids = tuple(str(item.parent_asin) for item in self._candidates)
+            terminal = response.terminal.termination_reason in {"awaiting_user", "failed"}
+            return (
+                self._observation(
+                    candidate_ids=ids,
+                    note=f"route={response.route}; status={response.terminal.status}",
+                    terminal=terminal,
+                    needs_user_input=response.terminal.termination_reason == "awaiting_user",
+                ),
+                {
+                    "tool": "run_shopping_turn",
+                    "route": response.route,
+                    "candidate_count": len(ids),
+                    "terminal": response.terminal.termination_reason,
+                    "state_version": str((self.service.state_diagnostics(self.task_id) or {}).get("state_digest") or ""),
+                },
+                terminal,
+            )
+        if proposal.action is PlannerAction.RERANK:
+            ranking, _combined, _soft, diagnostics = self.service._m5_soft_rerank(
+                self.session_id,
+                self._candidates,
+                existing=(),
+            )
+            if ranking is not None:
+                ordered = ranking.by_identity()
+                self._candidates = tuple(
+                    sorted(self._candidates, key=lambda item: ordered[str(item.parent_asin)].reranked_rank)
+                )
+            return (
+                self._observation(
+                    candidate_ids=tuple(str(item.parent_asin) for item in self._candidates),
+                    reranked=True,
+                    note="existing M6.1 soft rerank",
+                ),
+                {"tool": "m6_1_soft_rerank", "applied": ranking is not None, "diagnostics": diagnostics},
+                False,
+            )
+        if proposal.action is PlannerAction.FINALIZE:
+            ids = tuple(str(item.parent_asin) for item in self._candidates)
+            return (
+                self._observation(candidate_ids=ids, terminal=True, reranked=True, note="finalized"),
+                {"tool": "finalize", "candidate_count": len(ids)},
+                True,
+            )
+        raise ValueError(f"unsupported planner action {proposal.action!r}")

+"""Structured HTTP Planner Action proposer (explicit opt-in, no fallback).
+
+This client asks an external HTTP model for one Planner action.  It never executes a
+tool itself; ``run_planner`` passes the returned ActionProposal through the existing
+contract validator and then the existing ServicePlannerExecutor.
+"""
+
+from __future__ import annotations
+
+import json
+import os
+import urllib.error
+import urllib.request
+from typing import Any
+
+from recommendation.demo.planner_contract import (
+    ActionProposal,
+    PlannerAction,
+    PlannerContractError,
+    PlannerObservation,
+)
+
+ENV_ENABLED = "AGENTRECX_QWEN_PLANNER"
+ENV_URL = "AGENTRECX_QWEN_PLANNER_URL"
+ENV_MODEL = "AGENTRECX_QWEN_PLANNER_MODEL"
+CONTRACT_VERSION = "agentrecx.planner_action.v1"
+
+
+def planner_enabled() -> bool:
+    return (os.environ.get(ENV_ENABLED, "").strip().casefold() in {"1", "true", "on", "yes", "enabled"})
+
+
+class HttpPlannerError(RuntimeError):
+    def __init__(self, code: str, detail: str) -> None:
+        super().__init__(detail)
+        self.code = code
+        self.detail = detail
+
+
+def parse_action_payload(payload: Any) -> ActionProposal:
+    if not isinstance(payload, dict):
+        raise HttpPlannerError("planner_payload_not_object", "planner response is not a JSON object")
+    if "operations" in payload:
+        # The Preference Operation schema is not a Planner Action schema.
+        raise HttpPlannerError("planner_used_preference_schema", "planner returned Preference Operation JSON")
+    raw_action = str(payload.get("action") or "").strip().upper()
+    try:
+        action = PlannerAction(raw_action)
+    except ValueError as exc:
+        raise HttpPlannerError("planner_invalid_action", f"unknown planner action {raw_action!r}") from exc
+    arguments = payload.get("arguments") or {}
+    if not isinstance(arguments, dict):
+        raise HttpPlannerError("planner_invalid_arguments", "planner arguments must be an object")
+    rationale = str(payload.get("rationale") or "")
+    return ActionProposal(action=action, arguments=dict(arguments), rationale=rationale)
+
+
+class HttpPlannerProposer:
+    def __init__(self, url: str, *, model: str | None = None, timeout: float = 120.0) -> None:
+        self.url = url.rstrip("/")
+        self.model = model
+        self.timeout = timeout
+
+    @classmethod
+    def from_env(cls) -> "HttpPlannerProposer":
+        if not planner_enabled():
+            raise HttpPlannerError("planner_disabled", "AGENTRECX_QWEN_PLANNER is not enabled")
+        url = (os.environ.get(ENV_URL) or "").strip()
+        if not url:
+            raise HttpPlannerError("planner_misconfigured", "AGENTRECX_QWEN_PLANNER_URL is empty")
+        return cls(url, model=(os.environ.get(ENV_MODEL) or "").strip() or None)
+
+    def propose(self, observation: PlannerObservation) -> ActionProposal:
+        request_payload = {
+            "contract_version": CONTRACT_VERSION,
+            "step": observation.step,
+            "query": observation.query,
+            "candidate_ids": list(observation.candidate_ids),
+            "candidate_count": len(observation.candidate_ids),
+            "state_version": observation.state_version,
+            "m5_state": observation.m5_state,
+            "allowed_actions": list(observation.allowed_actions),
+            "last_tool_observation": observation.last_tool_observation,
+            "remaining_steps": observation.remaining_steps,
+            "needs_user_input": observation.needs_user_input,
+            "reranked": observation.reranked,
+        }
+        body = json.dumps(request_payload, ensure_ascii=False).encode("utf-8")
+        request = urllib.request.Request(
+            self.url + "/plan",
+            data=body,
+            headers={"Content-Type": "application/json"},
+            method="POST",
+        )
+        try:
+            with urllib.request.urlopen(request, timeout=self.timeout) as response:
+                payload = json.loads(response.read().decode("utf-8"))
+        except (urllib.error.URLError, TimeoutError, OSError) as exc:
+            raise HttpPlannerError("planner_unavailable", "the planner model server is unavailable") from exc
+        except (ValueError, UnicodeDecodeError) as exc:
+            raise HttpPlannerError("planner_invalid_json", "the planner model server returned invalid JSON") from exc
+        return parse_action_payload(payload)
