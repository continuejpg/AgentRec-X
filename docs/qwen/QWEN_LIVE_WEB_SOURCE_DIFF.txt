+### Configuration
+
+Set these in the **same Windows PowerShell session** that launches the demo (the launcher forwards
+them into WSL; the endpoint is never shown in the browser):
+
+```powershell
+$env:QWEN_LIVE_ENABLED = "1"                       # explicit opt-in
+$env:QWEN_MODEL_ENDPOINT = "http://127.0.0.1:18009"  # existing model server, NOT auto-started
+$env:QWEN_MODEL_NAME = "Qwen3.5-9B-Base"
+$env:QWEN_LIVE_TIMEOUT_S = "180"                    # each /plan call
+$env:QWEN_LIVE_HEALTH_TIMEOUT_S = "60"             # real PlannerAction health probe
+$env:QWEN_LIVE_TURN_TIMEOUT_S = "600"              # bounds the complete multi-step turn
+$env:QWEN_LIVE_MAX_STEPS = "8"
+powershell -ExecutionPolicy Bypass -File scripts/run_web_demo.ps1
+```
+
+Availability is deliberately strict: the BFF performs a real `POST /plan` health probe and only
+shows Qwen Live as available after it parses a valid PlannerAction. A configured-but-unreachable
+service leaves **Qwen Live Unavailable**; CPU Live and Qwen Replay continue to work, and a Qwen
+request never silently becomes a Rule request. The BFF does **not** start/stop AutoDL/GPU, create
+SSH tunnels, download or quantize a model. Start the model service yourself, using the repository's
+`training/agent_sft/model_server.py` or your existing endpoint.

+The MCP server exposes the original 7 tools plus one conditional 8th tool,
+`run_agent_turn`, only when Qwen Live is configured. `run_agent_turn` returns the model's real
+actions, tool observations, Gate result, P15 evidence, planner/explainer latencies and the
+optional Qwen clarification handle. `/api/chat/stream` accepts
+`{"mode":"qwen_live", ...}` and emits the same SSE event names plus planner metadata on `done`.

+class _ProgressBridge:
+    """Small event bridge from the synchronous executor/Qwen calls to MCP progress notifications.
+
+    It does not own any Agent state: the existing ``run_planner`` + ``ElectronicsPlannerExecutor``
+    are still the only loop/executor.  Events carry action/tool names, status, counts and real
+    latency only; prompts and hidden reasoning are never sent.
+    """
+
+    def __init__(self, context: Context, loop: asyncio.AbstractEventLoop) -> None:
+        self.context = context
+        self.loop = loop
+        self.counter = 0
+        self.pending: list[asyncio.Future[Any]] = []
+
+    def emit(self, *, tool: str, label: str, status: str,
+             detail: str | None = None, latency_ms: float | None = None) -> None:
+        self.counter = min(99, self.counter + 1)
+        payload: dict[str, Any] = {
+            "tool": tool, "label": label, "status": status, "progress": self.counter,
+        }
+        if detail is not None:
+            payload["detail"] = detail
+        if latency_ms is not None:
+            payload["latency_ms"] = round(float(latency_ms), 2)
+        try:
+            future = asyncio.run_coroutine_threadsafe(self._send(payload), self.loop)
+            self.pending.append(future)
+            future.add_done_callback(lambda done: done.exception())
+        except Exception:  # progress is observability; it must never fail the real execution
+            LOGGER.debug("progress notification could not be scheduled", exc_info=True)
+
+    async def flush(self) -> None:
+        """Wait until every queued progress notification has been handed to the transport."""
+        while self.pending:
+            future = self.pending.pop(0)
+            try:
+                await asyncio.wrap_future(future)
+            except Exception:
+                LOGGER.debug("queued progress notification failed", exc_info=True)
+
+    async def _send(self, payload: Mapping[str, Any]) -> None:
+        try:
+            await self.context.report_progress(
+                float(payload.get("progress") or 0), 100.0,
+                json.dumps(payload, ensure_ascii=False, separators=(",", ":")),
+            )
+        except Exception:  # no progress token / disconnected client: execution continues
+            LOGGER.debug("progress notification could not be delivered", exc_info=True)

+/**
+ * Merges one real `tool_status` event into the timeline.
+ *
+ * `running` opens a row (a repeated identical notification is not duplicated). A terminal status
+ * closes the newest matching running row; rows are matched on tool *and* label first because a
+ * Qwen turn legitimately repeats the same tool with a different label per planner action
+ * (`planner_action` → "Planner Action: RETRIEVE", then "Planner Action: RERANK", …). If the
+ * backend only ever sent a terminal status, it is appended as its own completed row — never
+ * dropped, never turned into a fake animation.
+ */
+export function mergeToolStatus(list: LocalToolStatus[], incoming: ToolStatus): LocalToolStatus[] {
+  if (incoming.status === "running") {
+    const last = list[list.length - 1];
+    if (last && last.status === "running" && last.tool === incoming.tool && last.label === incoming.label) {
+      // MCP progress notifications may repeat; refresh the open row instead of duplicating it.
+      return replaceRow(list, list.length - 1, { ...incoming, receivedAt: performance.now() });
+    }
+    return [...list, { ...incoming, receivedAt: performance.now() }];
+  }
+  let fallbackIndex = -1;
+  for (let index = list.length - 1; index >= 0; index -= 1) {
+    const row = list[index];
+    if (row.tool !== incoming.tool || row.status !== "running") continue;
+    if (row.label === incoming.label) return replaceRow(list, index, incoming);
+    if (fallbackIndex < 0) fallbackIndex = index;
+  }
+  if (fallbackIndex >= 0) return replaceRow(list, fallbackIndex, incoming);
+  return [...list, { ...incoming }];
+}

+ * explicit note — never as an endless spinner (fake animation) and never as a fabricated "ok".
