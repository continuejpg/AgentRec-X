+"""P6.3 minimal read-only runner: execute the six P6 dynamic DEV tasks through the
+real deterministic Rule Agent and score the final presented products.
+
+What it does
+------------
+1. Composes the **real** demo runtime (Serving V1 Fusion engine + real catalogue +
+   DemoAgentService + M5 decision state) with a fixed, documented configuration.
+2. Creates one isolated demo session per task from a fixed server-owned profile.
+3. Sends the task's turns verbatim to ``DemoAgentService.run_shopping_turn`` - the same
+   method the HTTP shopping route calls.
+4. Records the real final ``response.recommendations`` (post promotion), per-turn
+   tool/source calls, terminal state, M5 constraints, role diagnostics and memory state.
+5. Derives the task's expected final shopping state from the task file itself (initial
+   request + turn commands) and independently checks the catalogue for an eligible
+   product.  No Agent output and no gold product identity is used to build the
+   expectation.
+6. Produces per-task PASS / FAIL / UNKNOWN plus the six P6 dynamic metric inputs.
+
+It does not modify the Agent, eligibility, role rules, serving profile, tasks or
+evaluator.  It does not run Qwen, the holdout split, Fusion-20k or any training.
+"""
+

+DEFAULT_OUT = REPO / "runs" / "p6_3_dynamic"
+P6_DIR = REPO / "data" / "p6_benchmark_v0"
+PRODUCTS = REPO / "data" / "processed" / "Sports_and_Outdoors_products.jsonl"
+SEQUENCES = REPO / "data" / "processed" / "Sports_and_Outdoors_sequences.json"
+MAPPINGS = REPO / "data" / "processed" / "Sports_and_Outdoors_mappings.json"
+CHECKPOINT = REPO / "runs" / "sasrec_canonical_2026" / "best.pt"

+#: dynamic tasks explicitly exercise M5 state updates; the rest is the accepted deterministic
+#: Agent configuration (deterministic policy, history + catalog_search, k=5, default limits).

+SERVING_FUSION = "rrf3"

+PRICE_RE = re.compile(r"\bunder\s*\$?\s*(\d+(?:\.\d+)?)", re.IGNORECASE)
+UPDATE_PRICE_RE = re.compile(r"update\s+price_max\s+to\s*(\d+(?:\.\d+)?)", re.IGNORECASE)
+GOAL_SWITCH_RE = re.compile(r"switch\s+goal\s+to\s+(.+)", re.IGNORECASE)

+def _unique_dir(base: Path) -> Path:
+    if not base.exists():
+        return base
+    for suffix in range(2, 100):
+        candidate = base.with_name(f"{base.name}_r{suffix}")
+        if not candidate.exists():
+            return candidate
+    raise RuntimeError(f"no free output directory next to {base}")

+def load_tasks(path: Path) -> list[dict[str, Any]]:
+    tasks = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line.strip()]
+    if [t.get("split") for t in tasks] != ["dev"] * len(tasks):
+        raise ValueError("P6.3 only runs the dynamic DEV split")
+    if len(tasks) != 6:
+        raise ValueError(f"expected 6 dynamic dev tasks, got {len(tasks)}")
+    return tasks
+

+def _price_of(record: Any) -> float | None:
+    if record is None:
+        return None
+    for attribute in ("price_text", "price"):
+        value = getattr(record, attribute, None)
+        if value is None:
+            continue
+        match = re.search(r"(\d+(?:\.\d+)?)", str(value).replace(",", ""))
+        if match:
+            return float(match.group(1))
+    return None
+

+def expected_from_task(task: Mapping[str, Any]) -> dict[str, Any]:
+    """Derive the task's expected final state from its own request + turn commands.
+
+    This is a scripted interpretation of the *task definitions* (what each turn asks to
+    change), not a gold product identity and not a reading of the Agent's output.
+    """
+    from recommendation.demo.product_role import ProductTarget, TargetRole, infer_target
+
+    initial = str(task["initial_request"])
+    target = infer_target(initial)
+    price_match = PRICE_RE.search(initial)
+    price_max: float | None = float(price_match.group(1)) if price_match else None
+    weight_active = "lighter" in initial.casefold()
+    state_after_turn: list[dict[str, Any]] = [
+        {"turn": 0, "message": initial, "price_max": price_max,
+         "target_type": target.type, "target_role": target.role.value, "weight_active": weight_active}
+    ]
+    for index, turn in enumerate(task["turns"][1:], start=1):
+        message = str(turn["message"])
+        lowered = message.casefold()
+        if match := UPDATE_PRICE_RE.search(message):
+            price_max = float(match.group(1))
+        elif "remove budget" in lowered:
+            price_max = None
+        elif "retract" in lowered and "lighter" in lowered:
+            weight_active = False
+        elif match := GOAL_SWITCH_RE.search(message):
+            switched = infer_target(match.group(1))
+            if switched.type is None:
+                raise ValueError(f"goal switch in {task['task_id']} has no supported product type")
+            target = ProductTarget(switched.type, TargetRole.MAIN)
+        elif "replacement accessory role" in lowered:
+            if target.type is None:
+                raise ValueError(f"accessory role in {task['task_id']} has no product type")
+            target = ProductTarget(target.type, TargetRole.ACCESSORY)
+        state_after_turn.append(
+            {
+                "turn": index,
+                "message": message,
+                "price_max": price_max,
+                "target_type": target.type,
+                "target_role": target.role.value,
+                "weight_active": weight_active,
+            }
+        )
+    return {
+        "task_id": task["task_id"],
+        "template_id": task["template_id"],
+        "target": target,
+        "price_max": price_max,
+        "weight_active": weight_active,
+        "state_after_turn": state_after_turn,
+        "checks": dict(task.get("checks") or {}),
+        "task": task,
+    }
+
+
+def catalogue_evidence(records: Mapping[str, Any], target: Any, price_max: float | None) -> dict[str, Any]:
+    """Independent scan of the trusted catalogue for a product satisfying the task goal.
+
+    Only the catalogue record and the task-derived target/constraint are used; no Agent
+    candidate and no gold identity enters this function.
+    """
+    from recommendation.demo.product_role import RoleStatus, assess_candidate
+
+    eligible = 0
+    role_match = 0
+    role_unknown = 0
+    price_unknown = 0
+    examples: list[str] = []
+    for parent_asin, record in records.items():
+        assessment = assess_candidate(str(parent_asin), record, target)
+        if assessment.status is RoleStatus.MISMATCH:
+            continue
+        if assessment.status is RoleStatus.UNKNOWN:
+            role_unknown += 1
+            continue
+        role_match += 1
+        if price_max is None:
+            eligible += 1
+            if len(examples) < 3:
+                examples.append(str(parent_asin))
+            continue
+        price = _price_of(record)
+        if price is None:
+            price_unknown += 1
+            continue
+        if price <= price_max:
+            eligible += 1
+            if len(examples) < 3:
+                examples.append(str(parent_asin))
+    return {
+        "catalogue_size": len(records),
+        "eligible_verified_count": eligible,
+        "role_match_count": role_match,
+        "role_unknown_count": role_unknown,
+        "eligible_price_unknown_count": price_unknown,
+        "example_eligible_asins": examples,
+    }
+
+
+def _price_from_requirements(requirements: Sequence[Mapping[str, Any]]) -> list[float]:
+    values: list[float] = []
+    for requirement in requirements or ():
+        if str(requirement.get("kind")) != "price_max":
+            continue
+        try:
+            values.append(float(str(requirement.get("expected"))))
+        except (TypeError, ValueError):
+            continue
+    return values
+
+
+def _price_from_constraint_trace(trace: Mapping[str, Any] | None) -> list[float]:
+    if not trace:
+        return []
+    values: list[float] = []
+    for pair in trace.get("final_constraints") or ():
+        if not isinstance(pair, (list, tuple)) or len(pair) != 2:
+            continue
+        if str(pair[0]) != "price_max":
+            continue
+        try:
+            values.append(float(str(pair[1])))
+        except (TypeError, ValueError):
+            continue
+    return values
+
+
+def _active_weight(state: Any) -> bool:
+    entries = tuple(getattr(state, "active_entries", ()) or ())
+    for entry in entries:
+        kind = getattr(getattr(entry, "kind", None), "value", getattr(entry, "kind", None))
+        if str(kind) == "weight":
+            return True
+    return False
+
+
+def _role_target_from_diagnostics(diagnostics: Mapping[str, Any] | None) -> dict[str, Any] | None:
+    if not diagnostics:
+        return None
+    target = diagnostics.get("target") or diagnostics.get("role_target")
+    return dict(target) if isinstance(target, Mapping) else None
+
+
+def _role_statuses(final_ids: Sequence[str], records: Mapping[str, Any], target: Any) -> list[dict[str, Any]]:
+    from recommendation.demo.product_role import assess_candidate
+
+    out: list[dict[str, Any]] = []
+    for identity in final_ids:
+        record = records.get(identity)
+        assessment = assess_candidate(identity, record, target)
+        out.append(
+            {
+                "parent_asin": identity,
+                "status": assessment.status.value,
+                "reasons": list(assessment.reasons),
+                "title": getattr(record, "title", None) if record is not None else None,
+                "price": _price_of(record),
+            }
+        )
+    return out
+
+
+def _diagnose(
+    *,
+    expected: Mapping[str, Any],
+    turns: Sequence[Mapping[str, Any]],
+    final_body: Mapping[str, Any] | None,
+    final_diagnostics: Mapping[str, Any] | None,
+    final_role_diagnostics: Mapping[str, Any] | None,
+    final_candidate_filter: Mapping[str, Any] | None,
+    final_memory: Any,
+    records: Mapping[str, Any],
+) -> dict[str, Any]:
+    """Score one task from real outputs and the independent task-derived expectation."""
+    target = expected["target"]
+    checks = expected["checks"]
+    errors: list[str] = []
+    unverified: list[str] = []
+
+    if final_body is None or final_diagnostics is None:
+        return {
+            "result": "UNKNOWN",
+            "verified_success": False,
+            "unknown": True,
+            "reasons": ["no final Agent response or M5 diagnostics"],
+            "tool_calls": sum(int((turn.get("observation") or {}).get("tool_calls_used", 0)) for turn in turns),
+            "role_correct": 0, "role_total": 0, "role_evidence": [],
+            "hard_constraint_violations": 0,
+            "preference_update_consistent": False,
+            "eligible_candidate_present": False,
+            "final_ids": [],
+        }
+
+    final_ids = [str(item.get("parent_asin", "")) for item in final_body.get("recommendations", ()) or ()]
+    role_evidence = _role_statuses(final_ids, records, target)
+    role_correct = sum(1 for row in role_evidence if row["status"] == "MATCH")
+    role_mismatch = sum(1 for row in role_evidence if row["status"] == "MISMATCH")
+    role_unknown = sum(1 for row in role_evidence if row["status"] == "UNKNOWN")
+    if role_mismatch:
+        errors.append(f"{role_mismatch} final product(s) role-MISMATCH")
+    if role_unknown:
+        unverified.append(f"{role_unknown} final product(s) role-UNKNOWN")
+
+    expected_price = expected["price_max"]
+    price_violations: list[str] = []
+    price_unknown: list[str] = []
+    for row in role_evidence:
+        if row["price"] is None:
+            price_unknown.append(row["parent_asin"])
+        elif expected_price is not None and row["price"] > expected_price + 1e-9:
+            price_violations.append(f"{row['parent_asin']}:{row['price']}>{expected_price}")
+    if price_violations:
+        errors.append("hard-constraint violation: " + ", ".join(price_violations))
+    if price_unknown and expected_price is not None:
+        unverified.append("final product price unknown for: " + ", ".join(price_unknown))
+
+    # Per-turn M5 state consistency.  Only the dimensions a task actually asks to change
+    # are scored: an ordinary task is not failed because an unrelated goal/role diagnostic
+    # is absent, while a goal-switch/accessory task is UNKNOWN when that evidence is missing.
+    messages = [str(item.get("message", "")).casefold() for item in turns]
+    price_relevant = any(
+        state["price_max"] is not None or "budget" in message
+        for state, message in zip(expected["state_after_turn"], messages)
+    )
+    weight_relevant = any(state["weight_active"] for state in expected["state_after_turn"])
+    target_relevant = any(
+        ("switch goal" in message) or ("replacement accessory role" in message)
+        for message in messages
+    )
+    state_checks: list[dict[str, Any]] = []
+    for expected_state, turn in zip(expected["state_after_turn"], turns):
+        trace = turn.get("constraint_trace") or {}
+        actual_prices = _price_from_constraint_trace(trace)
+        expected_prices = [] if expected_state["price_max"] is None else [expected_state["price_max"]]
+        price_ok = sorted(actual_prices) == sorted(expected_prices) if price_relevant else None
+        role_diag = turn.get("role_diagnostics") or {}
+        actual_target = _role_target_from_diagnostics(role_diag)
+        if not target_relevant or expected_state["target_type"] is None:
+            target_ok: bool | None = None
+        elif actual_target is None:
+            target_ok = None
+        else:
+            target_ok = (
+                str(actual_target.get("type")) == str(expected_state["target_type"])
+                and str(actual_target.get("role")) == str(expected_state["target_role"])
+            )
+        actual_weight = turn.get("weight_active")
+        weight_ok = (
+            actual_weight == expected_state["weight_active"]
+            if weight_relevant and actual_weight is not None
+            else None
+        )
+        state_ok = not any(value is False for value in (price_ok, target_ok, weight_ok))
+        state_checks.append(
+            {
+                "turn": expected_state["turn"],
+                "message": expected_state["message"],
+                "expected": expected_state,
+                "actual_price_max": actual_prices,
+                "actual_target": actual_target,
+                "actual_weight_active": actual_weight,
+                "price_relevant": price_relevant,
+                "target_relevant": target_relevant,
+                "weight_relevant": weight_relevant,
+                "price_ok": price_ok,
+                "target_ok": target_ok,
+                "weight_ok": weight_ok,
+                "state_ok": state_ok,
+            }
+        )
+    def _has_failure(item: Mapping[str, Any]) -> bool:
+        return any(
+            item[key] is False for key in ("price_ok", "target_ok", "weight_ok")
+        )
+
+    def _has_missing_evidence(item: Mapping[str, Any]) -> bool:
+        return bool(
+            (item["price_ok"] is None and price_relevant)
+            or (item["target_ok"] is None and target_relevant)
+            or (item["weight_ok"] is None and weight_relevant)
+        )
+
+    state_fail = [item for item in state_checks if _has_failure(item)]
+    state_unknown = [
+        item for item in state_checks if not _has_failure(item) and _has_missing_evidence(item)
+    ]
+    if state_fail:
+        errors.append("per-turn M5 state mismatch at turn(s) " + ",".join(str(i["turn"]) for i in state_fail))
+    if state_unknown:
+        unverified.append("per-turn M5 state evidence incomplete at turn(s) " + ",".join(str(i["turn"]) for i in state_unknown))
+    preference_consistent = not state_fail and not state_unknown
+
+    catalogue = expected["catalogue_evidence"]
+    eligible_exists = catalogue["eligible_verified_count"] > 0
+    # "No eligible product" is only a verified statement when the catalogue scan can decide
+    # the price of every role-matching product.  Otherwise the task outcome is UNKNOWN, not
+    # a PASS for reporting no match and not a FAIL for the Agent missing a product.
+    catalogue_existence_verified = eligible_exists or (
+        catalogue["role_match_count"] == 0
+        and catalogue["eligible_price_unknown_count"] == 0
+        and catalogue["role_unknown_count"] == 0
+    )
+    eligible_present = any(row["status"] == "MATCH" for row in role_evidence) and (
+        expected_price is None
+        or any(
+            row["status"] == "MATCH" and row["price"] is not None and row["price"] <= expected_price + 1e-9
+            for row in role_evidence
+        )
+    )
+    terminal = final_body.get("terminal") or {}
+    termination_ok = str(terminal.get("status")) == "finished" and bool(
+        final_body.get("recommendations")
+    )
+
+    # Completion / presence rules.
+    if checks.get("must_report_unknown_when_no_eligible") and not eligible_exists:
+        if not catalogue_existence_verified:
+            unverified.append(
+                "catalogue eligibility cannot be decided: "
+                f"{catalogue['role_match_count']} role-matching product(s), "
+                f"{catalogue['eligible_price_unknown_count']} with unknown price"
+            )
+        if final_ids:
+            errors.append("catalogue has no verified eligible product but the Agent presented products")
+        else:
+            message = str(final_body.get("message") or "").casefold()
+            reports_absence = any(
+                phrase in message
+                for phrase in ("no ", "none", "couldn't", "cannot", "unable", "not find", "no matching",
+                               "no suitable", "no product", "no item", "budget")
+            )
+            if not reports_absence:
+                unverified.append("catalogue has no verified eligible product and the Agent did not report it")
+    else:
+        if not final_ids:
+            if eligible_exists:
+                errors.append("catalogue contains verified eligible products but the Agent presented none")
+            else:
+                unverified.append("no final products and no verified catalogue-eligible product")
+        elif not termination_ok:
+            unverified.append(f"final terminal status is {terminal.get('status')!r}, recommendations={len(final_ids)}")
+
+    if errors:
+        result = "FAIL"
+    elif unverified:
+        result = "UNKNOWN"
+    else:
+        result = "PASS"
+
+    return {
+        "result": result,
+        "verified_success": result == "PASS",
+        "unknown": result == "UNKNOWN",
+        "reasons": errors + unverified,
+        "tool_calls": sum(int((turn.get("observation") or {}).get("tool_calls_used", 0)) for turn in turns),
+        "role_correct": role_correct,
+        "role_total": len(role_evidence),
+        "role_evidence": role_evidence,
+        "hard_constraint_violations": len(price_violations),
+        "hard_constraint_violation_detail": price_violations,
+        "preference_update_consistent": preference_consistent,
+        "per_turn_state": state_checks,
+        "eligible_candidate_present": eligible_present,
+        "eligible_catalogue_verified": eligible_exists,
+        "catalogue_existence_verified": catalogue_existence_verified,
+        "catalogue_evidence": catalogue,
+        "final_ids": final_ids,
+        "final_terminal": dict(terminal),
+        "final_requirements": list(final_body.get("eligibility", {}).get("requirements", ()) or ()),
+        "final_eligibility": final_body.get("eligibility"),
+        "final_candidate_filter": final_candidate_filter,
+        "final_role_diagnostics": final_role_diagnostics,
+        "final_m5_diagnostics": final_diagnostics,
+        "final_memory_weight_active": _active_weight(final_memory) if final_memory is not None else None,
+    }
+
+
+def main(argv: Sequence[str] | None = None) -> int:
+    parser = argparse.ArgumentParser(description="P6.3 real Rule Agent dynamic-dev runner")
+    parser.add_argument("--tasks", type=Path, default=DEFAULT_TASKS)
+    parser.add_argument("--out", type=Path, default=None)
+    parser.add_argument("--profile", default=PROFILE_ID)
+    parser.add_argument("--k", type=int, default=DEFAULT_K)
+    args = parser.parse_args(argv)
+
+    tasks = load_tasks(args.tasks)
+    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
+    out = _unique_dir(args.out or (DEFAULT_OUT / f"dev_{timestamp}"))
+    out.mkdir(parents=True, exist_ok=False)
+
+    # Fixed run configuration; set before ServiceSettings / state integration are read.
+    os.environ["AGENTRECX_STATE_MODE"] = STATE_MODE
+    os.environ["AGENTRECX_STATE_DB"] = str(out / "decision_state.sqlite3")
+    os.environ["AGENTRECX_MEMORY_DB"] = str(out / "preference_memory.sqlite3")
+    os.environ["AGENTRECX_SERVING_FUSION"] = SERVING_FUSION
+    os.environ["AGENTRECX_CATALOG_METADATA_PATH"] = str(PRODUCTS)
+    os.environ.pop("AGENTRECX_QWEN_PROPOSAL", None)
+    os.environ.pop("AGENTRECX_AGENT_POLICY", None)
+
+    from recommendation.api.app import ServiceSettings, build_engine
+    from recommendation.demo.agent_schemas import AgentRecommendRequest
+    from recommendation.demo.runtime import build_demo_runtime
+    from experiments.p6_benchmark.metrics import dynamic_metrics
+
+    started = time.perf_counter()
+    settings = ServiceSettings.from_env()
+    engine, engine_error = build_engine(settings)
+    if engine is None:
+        raise SystemExit(f"serving engine could not be built: {engine_error}")
+    engine_meta = engine.model_metadata()
+    if engine_meta.get("active_engine") != "fusion_rrf":
+        raise SystemExit(f"expected fusion_rrf, got {engine_meta.get('active_engine')!r}")
+    runtime = build_demo_runtime(settings=settings, engine=engine, memory_db=out / "preference_memory.sqlite3")
+    service = runtime.agent_service()
+    if service.policy_name != "deterministic":
+        raise SystemExit(f"expected deterministic Rule Agent, got policy {service.policy_name!r}")
+    if args.profile not in runtime.profiles:
+        raise SystemExit(f"profile {args.profile!r} not in runtime profiles")
+    profile = runtime.profiles[args.profile]
+    records = runtime.metadata.records
+
+    turns_rows: list[dict[str, Any]] = []
+    task_rows: list[dict[str, Any]] = []
+    metric_rows: list[dict[str, Any]] = []
+    for task in tasks:
+        expected = expected_from_task(task)
+        expected["catalogue_evidence"] = catalogue_evidence(records, expected["target"], expected["price_max"])
+        session_id = runtime.sessions.create(args.profile).session_id
+        session = runtime.sessions.get(session_id)
+        task_key = session.session_id
+        user_key = session.user_key
+        turn_records: list[dict[str, Any]] = []
+        final_body: dict[str, Any] | None = None
+        execution_error: str | None = None
+        for index, turn in enumerate(task["turns"]):
+            message = str(turn["message"])
+            try:
+                with runtime.sessions.turn(session_id) as allocation:
+                    session = allocation.session
+                    payload = AgentRecommendRequest(
+                        message=message,
+                        history=list(session.trusted_user_history),
+                        k=args.k,
+                        sources=DEFAULT_SOURCES,
+                        user_key=session.user_key,
+                    )
+                    turn_started = time.perf_counter()
+                    response = service.run_shopping_turn(
+                        payload,
+                        session_id=session.user_key,
+                        task_id=session.session_id,
+                        turn_id=allocation.turn_id,
+                    )
+                    seconds = round(time.perf_counter() - turn_started, 3)
+                body = response.model_dump()
+                final_body = body
+                diagnostics = service.state_diagnostics(session.session_id)
+                trace = service.constraint_trace(session.session_id)
+                role_diag = service.role_diagnostics(session.user_key)
+                candidate_filter = service.candidate_filter_diagnostics(session.user_key)
+                memory = runtime.memory_service.get_active_preferences(session.user_key)
+                record = {
+                    "task_id": task["task_id"],
+                    "turn": index,
+                    "message": message,
+                    "seconds": seconds,
+                    "route": body.get("route"),
+                    "terminal": body.get("terminal"),
+                    "recommendations": body.get("recommendations"),
+                    "observation": body.get("observation"),
+                    "eligibility": body.get("eligibility"),
+                    "message_text": body.get("message"),
+                    "constraint_trace": trace,
+                    "state_diagnostics": diagnostics,
+                    "role_diagnostics": role_diag,
+                    "candidate_filter": candidate_filter,
+                    "weight_active": _active_weight(memory),
+                }
+                turns_rows.append(record)
+                turn_records.append(record)
+            except Exception as exc:  # noqa: BLE001 - recorded as a real execution failure
+                execution_error = f"{type(exc).__name__}: {exc}"
+                record = {"task_id": task["task_id"], "turn": index, "message": message,
+                          "error": execution_error}
+                turns_rows.append(record)
+                turn_records.append(record)
+                break
+
+        diagnostics = service.state_diagnostics(task_key)
+        role_diag = service.role_diagnostics(user_key)
+        candidate_filter = service.candidate_filter_diagnostics(user_key)
+        memory = runtime.memory_service.get_active_preferences(user_key)
+        if execution_error:
+            evaluation = {
+                "result": "FAIL",
+                "verified_success": False,
+                "unknown": False,
+                "reasons": [f"execution error: {execution_error}"],
+                "tool_calls": sum(int((t.get("observation") or {}).get("tool_calls_used", 0)) for t in turn_records),
+                "role_correct": 0, "role_total": 0, "role_evidence": [],

+                final_diagnostics=diagnostics,
+                final_role_diagnostics=role_diag,
+                final_candidate_filter=candidate_filter,

+                "final_candidate_filter": candidate_filter,
+                "final_memory_weight_active": _active_weight(memory),
+                "evaluation": evaluation,
+            }

+            {
+                "verified_success": evaluation["verified_success"],
+                "hard_constraint_violations": evaluation["hard_constraint_violations"],
+                "role_correct": evaluation["role_correct"],
+                "role_total": evaluation["role_total"],
+                "preference_update_consistent": evaluation["preference_update_consistent"],
+                "eligible_candidate_present": evaluation["eligible_candidate_present"],
+                "tool_calls": evaluation["tool_calls"],
+                "unknown": evaluation["unknown"],
+            }
+        )
+        runtime.sessions.delete(session_id)
+
+    metrics = dynamic_metrics(metric_rows)
+    summary = {
+        "schema_version": "agentrecx.p6_3.dynamic_dev_run.v0",
+        "created_utc": _utc_now(),
+        "git_head": _git_head(),
+        "tasks_file": str(args.tasks.relative_to(REPO)) if args.tasks.is_relative_to(REPO) else str(args.tasks),

+            "source_head_depth": engine_meta.get("source_head_depth"),

+            "num_items": engine_meta.get("num_items"),
+        },
+        "resources": {
+            "device": "cpu",
+            "engine_and_runtime_build_seconds": round(time.perf_counter() - started, 1),
+            "wall_seconds": round(time.perf_counter() - started, 1),

+        "metrics": metrics,
+        "tasks": {row["task_id"]: row for row in task_rows},
